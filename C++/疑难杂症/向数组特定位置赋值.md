### 整体逻辑总结：

这段代码通过两个自定义函数（`assignToArray`和`printArray`）实现了 “向数组特定位置赋值” 和 “打印数组” 的功能，并在`main`函数中演示了正常赋值、越界赋值等场景。核心亮点是`assignToArray`函数的**边界检查**，能有效避免数组越界访问（C++ 中数组越界是常见错误，可能导致程序崩溃）。
```cpp
#include <iostream>
#include <iomanip>

// 函数：向数组特定位置赋值，包含边界检查
bool assignToArray(int arr[], int size, int index, int value) {
    // 检查索引是否在有效范围内
    if (index < 0 || index >= size) {
        std::cout << "错误：索引 " << index << " 超出数组范围 [0, " << size - 1 << "]" << std::endl;
        return false;
    }
    
    // 向特定位置赋值
    arr[index] = value;
    return true;
}

// 函数：打印数组内容
void printArray(int arr[], int size) {
    std::cout << "数组内容：";
    for (int i = 0; i < size; i++) {
        std::cout << std::setw(3) << arr[i];
    }
    std::cout << std::endl;
}

int main() {
    // 定义一个包含5个元素的数组
    const int ARRAY_SIZE = 5;
    int myArray[ARRAY_SIZE] = {10, 20, 30, 40, 50};
    
    std::cout << "初始状态：" << std::endl;
    printArray(myArray, ARRAY_SIZE);
    
    // 向特定位置赋值
    int position = 2;
    int newValue = 99;
    if (assignToArray(myArray, ARRAY_SIZE, position, newValue)) {
        std::cout << "\n向位置 " << position << " 赋值 " << newValue << " 后：" << std::endl;
        printArray(myArray, ARRAY_SIZE);
    }
    
    // 测试越界情况
    int invalidPosition = 10;
    std::cout << "\n尝试向位置 " << invalidPosition << " 赋值：" << std::endl;
    assignToArray(myArray, ARRAY_SIZE, invalidPosition, 100);
    
    // 再尝试几个赋值操作
    assignToArray(myArray, ARRAY_SIZE, 0, 100);
    assignToArray(myArray, ARRAY_SIZE, 4, 500);
    std::cout << "\n多次赋值后：" << std::endl;
    printArray(myArray, ARRAY_SIZE);
    
    return 0;
}
```
